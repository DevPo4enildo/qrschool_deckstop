```csharp
using Dapper;
using InventoryApp.Models;
using System.Collections.Generic;
using System.Linq;

namespace InventoryApp.DataAccess
{
    public class UserRepository
    {
        public User GetByUsername(string username)
        {
            using (var connection = DatabaseContext.CreateConnection())
            {
                // Теперь выбираем и поле password_plain
                var sql = "SELECT id, username, password_hash, password_plain, full_name, role, created_at, last_login FROM users WHERE username = @Username";
                return connection.QueryFirstOrDefault<User>(sql, new { Username = username });
            }
        }
        
        public void UpdateLastLogin(int userId)
        {
            using (var connection = DatabaseContext.CreateConnection())
            {
                var sql = "UPDATE users SET last_login = NOW() WHERE id = @UserId";
                connection.Execute(sql, new { UserId = userId });
            }
        }
        
        public IEnumerable<User> GetAll()
        {
            using (var connection = DatabaseContext.CreateConnection())
            {
                return connection.Query<User>("SELECT id, username, full_name, role, created_at, last_login FROM users ORDER BY id");
            }
        }
        
        // Новый метод для проверки пароля (сначала проверяем plain, потом hash)
        public bool ValidatePassword(User user, string password)
        {
            // 1. Проверяем по открытому паролю (для разработки)
            if (!string.IsNullOrEmpty(user.PasswordPlain) && user.PasswordPlain == password)
            {
                return true;
            }
            
            // 2. Если нет открытого пароля или не совпал, проверяем хеш
            if (!string.IsNullOrEmpty(user.PasswordHash))
            {
                var passwordHash = Helpers.PasswordHelper.HashPassword(password);
                return user.PasswordHash == passwordHash;
            }
            
            return false;
        }
        
        // Новый метод для создания пользователя с открытым паролем
        public int CreateUser(string username, string password, string fullName, string role)
        {
            using (var connection = DatabaseContext.CreateConnection())
            {
                var passwordHash = Helpers.PasswordHelper.HashPassword(password);
                var sql = @"
                    INSERT INTO users (username, password_hash, password_plain, full_name, role, created_at) 
                    VALUES (@Username, @PasswordHash, @PasswordPlain, @FullName, @Role, @CreatedAt) 
                    RETURNING id";
                
                return connection.QuerySingle<int>(sql, new
                {
                    Username = username,
                    PasswordHash = passwordHash,
                    PasswordPlain = password, // Сохраняем открытый пароль для разработки
                    FullName = fullName,
                    Role = role,
                    CreatedAt = System.DateTime.Now
                });
            }
        }
        
        // Обновление пароля
        public bool UpdatePassword(int userId, string newPassword)
        {
            using (var connection = DatabaseContext.CreateConnection())
            {
                var passwordHash = Helpers.PasswordHelper.HashPassword(newPassword);
                var sql = @"
                    UPDATE users 
                    SET password_hash = @PasswordHash, 
                        password_plain = @PasswordPlain 
                    WHERE id = @UserId";
                
                return connection.Execute(sql, new
                {
                    PasswordHash = passwordHash,
                    PasswordPlain = newPassword,
                    UserId = userId
                }) > 0;
            }
        }
    }
}
```

## 3. Обновленный AuthenticationService.cs

```csharp
using InventoryApp.DataAccess;
using InventoryApp.Models;
using System.Windows;

namespace InventoryApp.Services
{
    public class AuthenticationService
    {
        private readonly UserRepository _userRepository;
        public User CurrentUser { get; private set; }
        
        public AuthenticationService()
        {
            _userRepository = new UserRepository();
        }
        
        public bool Login(string username, string password)
        {
            // Получаем пользователя из БД
            var user = _userRepository.GetByUsername(username);
            if (user == null)
            {
                MessageBox.Show($"Пользователь '{username}' не найден", "Ошибка авторизации", 
                    MessageBoxButton.OK, MessageBoxImage.Warning);
                return false;
            }
            
            // Проверяем пароль
            if (!_userRepository.ValidatePassword(user, password))
            {
                MessageBox.Show("Неверный пароль", "Ошибка авторизации", 
                    MessageBoxButton.OK, MessageBoxImage.Warning);
                return false;
            }
            
            // Успешная авторизация
            CurrentUser = user;
            _userRepository.UpdateLastLogin(user.Id);
            
            // Для отладки показываем информацию о входе
            MessageBox.Show($"Добро пожаловать, {user.FullName}!\nРоль: {(user.Role == "accountant" ? "Бухгалтер" : "Технический специалист")}", 
                "Успешный вход", MessageBoxButton.OK, MessageBoxImage.Information);
            
            return true;
        }
        
        public void Logout()
        {
            CurrentUser = null;
        }
        
        public bool IsAccountant 
        { 
            get { return CurrentUser != null && CurrentUser.Role == "accountant"; } 
        }
        
        public bool IsTech 
        { 
            get { return CurrentUser != null && CurrentUser.Role == "tech"; } 
        }
        
        // Новый метод для отображения информации о пользователе
        public string GetUserInfo()
        {
            if (CurrentUser == null) return "Не авторизован";
            return $"{CurrentUser.FullName} ({CurrentUser.Role})";
        }
    }
}
```

## 5. Обновленное окно авторизации LoginWindow.xaml.cs

```csharp
using System.Windows;
using System.Windows.Controls;
using InventoryApp.Services;

namespace InventoryApp.Views
{
    public partial class LoginWindow : Window
    {
        private readonly AuthenticationService _authService;
        
        public LoginWindow()
        {
            InitializeComponent();
            _authService = new AuthenticationService();
            
            // Устанавливаем фокус на поле логина
            Loaded += (s, e) => txtUsername.Focus();
        }
        
        private void BtnLogin_Click(object sender, RoutedEventArgs e)
        {
            var username = txtUsername.Text.Trim();
            var password = txtPassword.Password;
            
            // Валидация
            if (string.IsNullOrEmpty(username))
            {
                MessageBox.Show("Введите логин", "Ошибка", 
                    MessageBoxButton.OK, MessageBoxImage.Warning);
                txtUsername.Focus();
                return;
            }
            
            if (string.IsNullOrEmpty(password))
            {
                MessageBox.Show("Введите пароль", "Ошибка", 
                    MessageBoxButton.OK, MessageBoxImage.Warning);
                txtPassword.Focus();
                return;
            }
            
            // Пытаемся авторизоваться
            if (_authService.Login(username, password))
            {
                // Открываем нужное окно в зависимости от роли
                if (_authService.IsAccountant)
                {
                    var accountantWindow = new AccountantInventoryWindow(_authService);
                    accountantWindow.Show();
                }
                else if (_authService.IsTech)
                {
                    var techWindow = new TechInventoryWindow(_authService);
                    techWindow.Show();
                }
                else
                {
                    MessageBox.Show("Неизвестная роль пользователя", "Ошибка", 
                        MessageBoxButton.OK, MessageBoxImage.Error);
                    return;
                }
                
                // Закрываем окно авторизации
                Close();
            }
            // Сообщения об ошибках выводятся внутри AuthenticationService
        }
        
        private void BtnExit_Click(object sender, RoutedEventArgs e)
        {
            var result = MessageBox.Show("Вы уверены, что хотите выйти?", "Подтверждение", 
                MessageBoxButton.YesNo, MessageBoxImage.Question);
            
            if (result == MessageBoxResult.Yes)
            {
                Application.Current.Shutdown();
            }
        }
        
        // Показать/скрыть пароль
        private void ChkShowPassword_Checked(object sender, RoutedEventArgs e)
        {
            // Создаем временное текстовое поле для отображения пароля
            if (txtPassword.Password.Length > 0)
            {
                var tempTextBox = new TextBox();
                tempTextBox.Text = txtPassword.Password;
                tempTextBox.PasswordChar = '\0';
                
                // Заменяем PasswordBox на TextBox
                var parent = txtPassword.Parent as Grid;
                if (parent != null)
                {
                    var row = Grid.GetRow(txtPassword);
                    var column = Grid.GetColumn(txtPassword);
                    parent.Children.Remove(txtPassword);
                    tempTextBox.Name = "txtPasswordVisible";
                    tempTextBox.FontSize = txtPassword.FontSize;
                    tempTextBox.Height = txtPassword.Height;
                    tempTextBox.Padding = txtPassword.Padding;
                    tempTextBox.Margin = txtPassword.Margin;
                    Grid.SetRow(tempTextBox, row);
                    Grid.SetColumn(tempTextBox, column);
                    parent.Children.Add(tempTextBox);
                    
                    // Сохраняем ссылку
                    txtPassword.Visibility = Visibility.Collapsed;
                }
            }
        }
        
        private void ChkShowPassword_Unchecked(object sender, RoutedEventArgs e)
        {
            // Возвращаем PasswordBox
            var parent = txtPassword.Parent as Grid;
            if (parent != null)
            {
                var visibleTextBox = parent.FindName("txtPasswordVisible") as TextBox;
                if (visibleTextBox != null)
                {
                    txtPassword.Password = visibleTextBox.Text;
                    parent.Children.Remove(visibleTextBox);
                    txtPassword.Visibility = Visibility.Visible;
                }
            }
        }
    }
}
```

## 8. Обновление PasswordHelper.cs (добавим метод для проверки пароля)

```csharp
using System;
using System.Security.Cryptography;
using System.Text;

namespace InventoryApp.Helpers
{
    public static class PasswordHelper
    {
        public static string HashPassword(string password)
        {
            using (var sha256 = SHA256.Create())
            {
                var hashedBytes = sha256.ComputeHash(Encoding.UTF8.GetBytes(password));
                return Convert.ToBase64String(hashedBytes);
            }
        }
        
        // Метод для верификации пароля (если нужно проверить только хеш)
        public static bool VerifyPassword(string password, string hash)
        {
            var hashOfInput = HashPassword(password);
            return hashOfInput == hash;
        }
    }
}
```

1. **Добавлено поле `PasswordPlain`** в модель User

3. **Улучшен AuthenticationService** - добавлены информационные сообщения при входе
4. **Переработан дизайн LoginWindow** - более современный вид, добавлена информация о тестовых учетках

6. **Улучшена валидация** полей ввода

Теперь система авторизации работает с обоими типами паролей - сначала проверяет открытый (для разработки), затем хешированный (для продакшена). Это очень удобно для тестирования!
